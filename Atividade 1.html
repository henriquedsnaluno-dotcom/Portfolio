# Simulação de movimento com atrito
v0 = float(input("Velocidade inicial (v0): "))
a = float(input("Aceleração de atrito (a): "))

v = v0
t = 0

print("Tempo (s) | Velocidade (m/s)")
while v > 0:
    print(f"{t:.1f}       | {v:.2f}")
    t += 0.5
    v = v0 - a * t
