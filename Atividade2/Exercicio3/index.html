# Lendo os dados do usuário velocidade_inicial = float(input("Digite a velocidade inicial (m/s): ")) aceleracao = float(input("Digite a aceleração (m/s²): ")) tempo = float(input("Digite o tempo (s): ")) # Calculando a velocidade final velocidade_final = velocidade_inicial + aceleracao * tempo # Calculando o espaço percorrido espaco_percorrido = velocidade_inicial * tempo + 0.5 * aceleracao * (tempo ** 2) # Exibindo os resultados print(f"Velocidade final: {velocidade_final:.2f} m/s") print(f"Espaço percorrido: {espaco_percorrido:.2f} m")
