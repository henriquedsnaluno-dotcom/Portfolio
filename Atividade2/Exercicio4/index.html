import math # Lê o diâmetro em polegadas diametro_polegadas = float(input("Digite o diâmetro do duto em polegadas: ")) # Converte para milímetros diametro_mm = diametro_polegadas * 25.4 # Calcula a área da seção transversal (A = π * r²) raio_mm = diametro_mm / 2 area_mm2 = math.pi * (raio_mm ** 2) # Exibe os resultados print(f"Diâmetro em milímetros: {diametro_mm:.2f} mm") print(f"Área da seção transversal: {area_mm2:.2f} mm²")
